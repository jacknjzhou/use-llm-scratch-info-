## 一、现状问题分析

### 1.1 当前实现架构

自动识别核心逻辑位于 `backend/app/services.py` 的 `classify_file` 函数：

```python
async def classify_file(file_row, schemas, llm_cfg):

    # 单次 LLM 调用

    # 返回结果

```

### 1.2 识别率低的根因

| 问题 | 描述 | 影响程度 |
|------|------|----------|
| **文本来源单一** | 仅前2页3000字符，复杂文档特征不足 | 高 |
| **Prompt 信息贫乏** | 只有模板名称/描述，缺少字段定义上下文 | 高 |
| **单次调用无验证** | 没有多角度校验或重试机制 | 中 |
| **阈值固定** | 静态阈值不适应不同文档场景 | 中 |
| **无历史复用** | 相同文件重复识别，浪费资源 | 低 |

### 3.1 问题

### 3.2 改进方案

**增加字段定义作为上下文**：

```python
async def classify_file_v2(file_row, schemas, llm_cfg, db):
```

```python
async def load_classification_examples(db: AsyncSession, schemas: list, limit: int = 5) -> str:
    """加载历史成功分类案例作为 few-shot 示例"""
    schema_ids = [str(s.id) for s in schemas]
    rows = (await db.execute(
        select(TaskFile)
        .where(TaskFile.match_status.in_(["bound", "need_confirm"]))
        .where(TaskFile.matched_schema_id.isnot(None))
        .order_by(TaskFile.task_id.desc())
        .limit(limit)
    )).scalars().all()
    
    if not rows:
        return ""
    
    file_ids = [r.file_id for r in rows]
    files = (await db.execute(
        select(UploadedFile).where(UploadedFile.id.in_(file_ids))
    )).scalars().all()
    file_map = {f.id: f for f in files}
    
    examples = ["参考以下历史分类案例："]
    for r in rows:
        f = file_map.get(r.file_id)
        if f:
            examples.append(f"- 文件「{f.filename}」→ 匹配「{r.matched_schema_id}」(置信度: {r.match_confidence})")
    
    return "\n".join(examples) + "\n"
```

## 四、阶段二：多页采样策略

### 4.1 问题

复杂文档（如长合同、多页报表）前两页可能没有核心特征，导致误判。

### 4.2 改进方案

**智能多页采样**：

```python
def sample_text_from_chunks(chunks: list[dict], max_chars: int = 5000, 
                            strategy: str = "head_middle_tail") -> str:
    """
    从分块中智能采样文本
    
    Args:
        chunks: parse_file 返回的分块列表
        max_chars: 最大字符数
        strategy: 采样策略
            - "head_middle_tail": 前中后各取一段
            - "content_rich": 优先选取内容密集的分块
            - "hybrid": 结合两种策略
    """
    if not chunks:
        return ""
    
    total_pages = max(c.get("page", 1) for c in chunks)
    
    if strategy == "head_middle_tail":
        # 前中后采样
        samples = []
        
        # 头部：前2页
        head_chunks = [c for c in chunks if c["page"] <= 2]
        for c in head_chunks:
            if sum(len(s) for s in samples) + len(c["text"]) <= max_chars // 3:
                samples.append(c["text"])
        
        # 中部：中间位置
        mid_page = max(2, total_pages // 2)
        mid_chunks = [c for c in chunks if abs(c["page"] - mid_page) <= 1]
        for c in mid_chunks[:2]:
            if sum(len(s) for s in samples) + len(c["text"]) <= max_chars // 3:
                samples.append(c["text"])
        
        # 尾部：最后1页
        tail_chunks = [c for c in chunks if c["page"] >= total_pages - 1]
        for c in tail_chunks[:2]:
            if sum(len(s) for s in samples) + len(c["text"]) <= max_chars // 3:
                samples.append(c["text"])
        
        return "\n...\n".join(samples)
    
    elif strategy == "content_rich":
        # 按内容密度排序，优先选取文字多的分块
        sorted_chunks = sorted(chunks, key=lambda c: len(c["text"]), reverse=True)
        samples = []
        for c in sorted_chunks:
            if sum(len(s) for s in samples) + len(c["text"]) <= max_chars:
                samples.append(c["text"])
            if sum(len(s) for s in samples) >= max_chars * 0.9:
                break
        return "\n".join(samples)
    
    else:  # hybrid
        samples = []
        head_text = "\n".join(c["text"] for c in chunks[:4] if c["page"] <= 2)
        samples.append(head_text)
        
        if len(chunks) > 4:
            sorted_by_len = sorted(chunks[4:], key=lambda c: len(c["text"]), reverse=True)
            if sorted_by_len:
                rich_text = sorted_by_len[0]["text"]
                if len(samples) + len(rich_text) <= max_chars:
                    samples.append(f"[中间关键页]\n{rich_text}")
        
        return "\n...\n".join(samples)
```

```python
async def classify_file_two_stage(file_row, schemas, llm_cfg, db) -> dict:
    """两阶段分类：粗筛 + 精排"""
    
    # 1. 多页采样
    chunks = parse_file(local_path, max_pages=settings.classify_max_pages)
    sampled_text = sample_text_from_chunks(chunks, 
                                           max_chars=settings.classify_max_chars,
                                           strategy=settings.classify_sample_strategy)
    
    # 2. 阶段一：粗筛
    coarse_candidates = await _coarse_classify(schemas, file_row.filename, sampled_text, llm_cfg)
    
    if not coarse_candidates:
        return {"schema_id": None, "confidence": 0.0, "reason": "粗筛无候选模板"}
    
    # 3. 阶段二：精排
    if len(coarse_candidates) == 1:
        result = coarse_candidates[0]
    else:
        result = await _fine_classify(coarse_candidates, file_row.filename, 
                                      sampled_text, llm_cfg, db)
    
    return result


async def _coarse_classify(schemas, filename, text, llm_cfg) -> list[dict]:
    """阶段一：粗筛 - 快速排除明显不匹配的模板"""
    
    template_list = "\n".join([
        f"{i+1}. {s.name} (类别: {s.category or '未分类'})"
        for i, s in enumerate(schemas)
    ])
    
    system = """你是一个文档分类助手。根据文件名和文档片段，快速判断该文档可能属于哪个模板类别。

输出要求：严格输出 JSON 数组，格式如下：
[
  {"index": 1, "name": "模板名", "confidence": 0.0~1.0},
  ...
]
- 返回置信度 > 0.3 的模板，最多返回 5 个
- 如果所有模板都不匹配，返回空数组 []
- 禁止编造，只输出你确定的判断
"""
    
    user = f"""文件名：{filename}

文档片段：
<<<
{text[:3000]}
>>>

模板列表：
{template_list}
"""
    
    data = await chat_json(llm_cfg.base_url, api_key, llm_cfg.model, system, user)
    
    try:
        candidates = json.loads(data) if isinstance(data, str) else data
        if not isinstance(candidates, list):
            return []
        
        schema_map = {s.name: s for s in schemas}
        results = []
        for item in candidates:
            name = item.get("name") or item.get("template_name")
            schema = schema_map.get(name)
            if schema:
                results.append({
                    "schema": schema,
                    "confidence": float(item.get("confidence", 0)),
                    "reason": f"粗筛匹配: {name}"
                })
        return sorted(results, key=lambda x: x["confidence"], reverse=True)[:5]
    except Exception:
        logger.warning("粗筛结果解析失败")
        return []


async def _fine_classify(candidates, filename, text, llm_cfg, db) -> dict:
    """阶段二：精排 - 详细分析候选模板"""
    
    candidate_detail = "\n".join([
        f"- id: {c['schema'].id}\n  name: {c['schema'].name}\n  category: {c['schema'].category or ''}\n  description: {c['schema'].description or ''}\n  key_fields: {json.dumps([{'key': f['key'], 'label': f.get('label', '')} for f in (c['schema'].fields or [])[:3]], ensure_ascii=False)}"
        for c in candidates
    ])
    
    schema_list = [c["schema"] for c in candidates]
    few_shot = await load_classification_examples(db, schema_list, limit=3)
    
    system = f"""你是专业的文档分类专家。根据给定的文档片段，从候选模板中选择最匹配的一个。

分类原则：
1. 仔细比对文档中的关键字段/特征词与模板字段定义的匹配度
2. 关注文档的结构特征（如合同条款格式、发票格式等）
3. 优先选择特征词匹配最多的模板

{few_shot}

输出要求：严格输出 JSON 对象：
{{"schema_id": "模板UUID", "confidence": 0.0~1.0, "reason": "详细判断理由", "matching_features": ["匹配特征1", "匹配特征2"]}}
"""
    
    user = f"""候选模板详情：
{candidate_detail}

文件名：{filename}

文档片段：
<<<
{text}
>>>
"""
    
    data = await chat_json(llm_cfg.base_url, api_key, llm_cfg.model, system, user)
    
    schema_id = data.get("schema_id")
    if schema_id:
        try:
            schema_id = uuid.UUID(str(schema_id))
        except ValueError:
            schema_id = None
    
    return {
        "schema_id": schema_id,
        "confidence": float(data.get("confidence") or 0),
        "reason": data.get("reason"),
        "matching_features": data.get("matching_features", [])
    }
```

## 六、阶段四：动态阈值 + 回退机制

### 6.1 问题

固定阈值无法适应不同文档场景，高置信度也可能误匹配。

### 6.2 改进方案

**动态阈值计算**：

```python
def calculate_dynamic_threshold(candidates: list[dict], base_threshold: float = 0.8) -> tuple[float, str]:
    """
    根据候选模板的分布动态调整阈值
    
    Returns:
        (threshold, strategy): 阈值和策略描述
    """
    if not candidates:
        return 0.5, "无候选，使用默认阈值"
    
    if len(candidates) == 1:
        return base_threshold * 0.9, "单一候选，降低阈值"
    
    top_conf = candidates[0]["confidence"]
    second_conf = candidates[1]["confidence"] if len(candidates) > 1 else 0
    gap = top_conf - second_conf
    
    if gap > 0.3:
        threshold = base_threshold * 1.1
        strategy = "候选差距大，提高阈值"
    elif gap < 0.1:
        threshold = base_threshold * 0.85
        strategy = "候选差距小，降低阈值"
    else:
        threshold = base_threshold
        strategy = "候选差距适中，使用默认阈值"
    
    return min(threshold, 0.95), strategy


async def classify_with_fallback(file_row, schemas, llm_cfg, db) -> dict:
    """带重试和回退的分类"""
    
    max_retries = 2
    last_error = None
    
    for attempt in range(max_retries):
        try:
            result = await classify_file_two_stage(file_row, schemas, llm_cfg, db)
            
            candidates = result.get("_candidates", [])
            threshold, strategy = calculate_dynamic_threshold(candidates)
            
            confidence = result["confidence"]
            result["threshold_strategy"] = strategy
            
            if result["schema_id"] is None or confidence < 0.3:
                result["match_status"] = "unmatched"
            elif confidence < threshold:
                result["match_status"] = "need_confirm"
            else:
                result["match_status"] = "bound"
            
            return result
            
        except Exception as exc:
            last_error = exc
            logger.warning(f"分类尝试 {attempt + 1} 失败: {exc}")
            
            if attempt < max_retries - 1:
                await asyncio.sleep(1)
                continue
    
    return {
        "schema_id": None,
        "confidence": 0,
        "reason": f"分类服务异常: {last_error}",
        "match_status": "need_confirm"  # 回退到需要确认
    }
```

## 七、阶段五：历史缓存 + 批量优化

### 7.1 问题

相同类型的文件重复识别，浪费 LLM 调用资源。

### 7.2 改进方案

**缓存机制设计**：

```python
# backend/app/services.py 新增

class MatchCache:
    """分类结果缓存"""
    
    def __init__(self, max_size: int = 1000, ttl_seconds: int = 86400):
        self._cache: dict[str, dict] = {}
        self._timestamps: dict[str, float] = {}
        self.max_size = max_size
        self.ttl_seconds = ttl_seconds
    
    def _make_key(self, filename: str, file_hash: str | None = None) -> str:
        import re
        normalized = re.sub(r'\d{4,}', 'N', filename)
        normalized = re.sub(r'V\d+', 'V', normalized)
        key = f"{normalized}:{file_hash or 'unknown'}"
        return key
    
    def get(self, filename: str, file_hash: str | None = None) -> dict | None:
        key = self._make_key(filename, file_hash)
        
        if key in self._timestamps:
            if time.time() - self._timestamps[key] > self.ttl_seconds:
                del self._cache[key]
                del self._timestamps[key]
                return None
        
        return self._cache.get(key)
    
    def set(self, filename: str, file_hash: str | None, result: dict):
        key = self._make_key(filename, file_hash)
        
        if len(self._cache) >= self.max_size and key not in self._cache:
            oldest_key = min(self._timestamps, key=self._timestamps.get)
            del self._cache[oldest_key]
            del self._timestamps[oldest_key]
        
        self._cache[key] = result
        self._timestamps[key] = time.time()
    
    def clear(self):
        self._cache.clear()
        self._timestamps.clear()


_match_cache = MatchCache()


async def classify_file_with_cache(file_row, schemas, llm_cfg, db, 
                                   file_hash: str | None = None) -> dict:
    """带缓存的分类"""
    
    cached = _match_cache.get(file_row.filename, file_hash)
    if cached:
        logger.info(f"分类缓存命中: {file_row.filename}")
        return {**cached, "from_cache": True}
    
    result = await classify_with_fallback(file_row, schemas, llm_cfg, db)
    
    if result.get("confidence", 0) >= 0.7 and result.get("schema_id"):
        _match_cache.set(file_row.filename, file_hash, result)
    
    return {**result, "from_cache": False}
```

**批量优化**：

```python
async def classify_batch_optimized(task_files: list[TaskFile], schemas: list, 
                                   llm_cfg: LlmConfig, db: AsyncSession) -> list[dict]:
    """批量分类优化：首个识别后批量应用"""
    
    results = []
    first_result = None
    
    pending_files = [tf for tf in task_files if tf.match_status == "pending"]
    
    if not pending_files:
        return results
    
    first_file = (await db.execute(
        select(UploadedFile).where(UploadedFile.id == pending_files[0].file_id)
    )).scalar_one()
    
    first_result = await classify_file_with_cache(first_file, schemas, llm_cfg, db)
    results.append({"file_id": str(first_file.id), **first_result})
    
    filename_pattern = normalize_filename(first_file.filename)
    
    for tf in pending_files[1:]:
        file_row = (await db.execute(
            select(UploadedFile).where(UploadedFile.id == tf.file_id)
        )).scalar_one()
        
        file_pattern = normalize_filename(file_row.filename)
        similarity = calculate_filename_similarity(filename_pattern, file_pattern)
        
        if similarity > 0.8 and first_result.get("confidence", 0) >= 0.85:
            results.append({
                "file_id": str(file_row.id),
                "schema_id": first_result["schema_id"],
                "confidence": first_result["confidence"] * 0.95,
                "reason": f"批量应用（文件名相似度: {similarity:.2f}）",
                "match_status": "bound" if first_result["confidence"] >= settings.match_auto_threshold else "need_confirm",
                "from_cache": False
            })
        else:
            file_result = await classify_file_with_cache(file_row, schemas, llm_cfg, db)
            results.append({"file_id": str(file_row.id), **file_result})
    
    return results


def normalize_filename(filename: str) -> str:
    import re
    normalized = re.sub(r'\d{4,}', 'N', filename)
    normalized = re.sub(r'V\d+', 'V', normalized)
    normalized = re.sub(r'[_\-\s]+', '', normalized).lower()
    return normalized


def calculate_filename_similarity(s1: str, s2: str) -> float:
    set1, set2 = set(s1), set(s2)
    if not set1 or not set2:
        return 0.0
    overlap = len(set1 & set2)
    return overlap / max(len(set1), len(set2))
```
